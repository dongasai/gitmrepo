### 3. 功能实现差异

#### pull 命令 - 缺少提交数量统计

**文档描述**:
```bash
✅ 已更新 (2 commits pulled)
```

**实际实现** ([src/commands/pull.ts:61-65](../src/commands/pull.ts#L61-L65)):
```typescript
if (output.includes('Already up to date') || output.includes('Already up-to-date')) {
  console.log('  ✅ 已经是最新的');
} else {
  console.log('  ✅ 已更新');  // 没有显示提交数量
}
```

**建议**: 增加 pull 后的提交数量统计，使用工具函数计算。

---

#### push 命令 - 推送逻辑问题

**问题**: 在 [src/commands/push.ts:62-71](../src/commands/push.ts#L62-L71) 中，无论推送成功或失败，都会在推送前显示未推送提交数量，推送后也显示同样的数量。

```typescript
const unpushedCount = countUnpushedCommits(fullPath, module.branch);

if (unpushedCount === 0) {
  console.log('  ✅ 没有需要推送的提交');
} else {
  console.log(`  未推送提交: ${unpushedCount}`);  // 推送前显示
}

try {
  execSync(`git push origin "${module.branch}"`, {
    cwd: fullPath,
    encoding: 'utf-8',
  });
  console.log(`  ✅ 已推送 ${unpushedCount} 个提交`);  // 推送后显示同样的数字
} catch (error) {
  // ...
}
```

**问题分析**:
1. 如果 `unpushedCount === 0`，代码会显示 "没有需要推送的提交" 然后继续执行 `git push`，这是无意义的操作
2. 推送失败时也会显示 "已推送 N 个提交"，因为 `console.log` 在 try 块中

**建议**: 修复逻辑：
- 如果 `unpushedCount === 0`，应该跳过推送操作
- 推送成功后再显示提交数量

---

#### 缺少分支信息显示

**文档描述**: 所有三个命令都显示 "分支: main"

**实际实现**: 三个命令都没有显示分支信息

**影响**: 用户无法直观看到当前操作的分支

**建议**: 在所有命令输出中增加分支信息显示。

## 实际代码示例

### pull 命令实际实现

```typescript
export async function pullExecute(moduleArg?: string): Promise<void> {
  const root = getGitRoot();
  const configPath = path.join(root, '.gitmrepo');

  if (!fs.existsSync(configPath)) {
    console.log('⚠️  .gitmrepo 配置文件不存在');
    console.log('   请先执行: git mrepo init');
    return;
  }

  const config = ConfigManager.load(configPath);
  const cm = new ConfigManager();

  // 确定模块列表
  const modules: Module[] = moduleArg
    ? [(() => {
        const m = cm.findModule(config, moduleArg);
        if (!m) throw new Error(`模块不存在: ${moduleArg}`);
        return m;
      })()]
    : Object.values(config.modules);

  console.log('🔄 拉取模块仓库更新...');

  for (const module of modules) {
    console.log(`[${module.name}] ${module.path}:`);

    const fullPath = path.isAbsolute(module.path) ? module.path : path.join(root, module.path);

    try {
      const output = execSync(`git pull origin "${module.branch}"`, {
        cwd: fullPath,
        encoding: 'utf-8',
      });

      if (output.includes('Already up to date') || output.includes('Already up-to-date')) {
        console.log('  ✅ 已经是最新的');
      } else {
        console.log('  ✅ 已更新');
      }
    } catch (error) {
      const stderr = (error as any).stderr?.toString() || '未知错误';
      console.log(`  ❌ ${stderr}`);
    }
  }
}
```

### push 命令实际实现

```typescript
export async function pushExecute(moduleArg?: string): Promise<void> {
  const root = getGitRoot();
  const configPath = path.join(root, '.gitmrepo');

  const config = ConfigManager.load(configPath);
  const cm = new ConfigManager();

  const modules: Module[] = moduleArg ? [...] : Object.values(config.modules);

  console.log('🔄 推送模块仓库变更...');

  for (const module of modules) {
    console.log(`[${module.name}] ${module.path}:`);

    const fullPath = path.join(root, module.path);
    const unpushedCount = countUnpushedCommits(fullPath, module.branch);

    if (unpushedCount === 0) {
      console.log('  ✅ 没有需要推送的提交');
    } else {
      console.log(`  未推送提交: ${unpushedCount}`);
    }

    try {
      execSync(`git push origin "${module.branch}"`, { cwd: fullPath });
      console.log(`  ✅ 已推送 ${unpushedCount} 个提交`);
    } catch (error) {
      console.log(`  ❌ ${stderr}`);
    }
  }
}
```

### fetch 命令实际实现

```typescript
export async function fetchExecute(moduleArg?: string): Promise<void> {
  const root = getGitRoot();
  const configPath = path.join(root, '.gitmrepo');

  const config = ConfigManager.load(configPath);
  const cm = new ConfigManager();

  const modules: Module[] = moduleArg ? [...] : Object.values(config.modules);

  console.log('🔄 获取模块仓库远程信息...');

  for (const module of modules) {
    console.log(`[${module.name}] ${module.path}:`);

    const fullPath = path.join(root, module.path);

    try {
      execSync('git fetch origin', { cwd: fullPath });
      console.log('  ✅ 已获取远程信息');

      const remoteNewCommits = countRemoteNewCommits(fullPath, module.branch);
      if (remoteNewCommits > 0) {
        console.log(`  远程新提交: ${remoteNewCommits}`);
      } else {
        console.log('  远程新提交: 0 (已经是最新)');
      }
    } catch (error) {
      console.log(`  ❌ ${stderr}`);
    }
  }
}
```
